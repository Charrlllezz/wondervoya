              console.log(`📍 Destination resolution: "${extractedDestination}" → "${targetDestination?.name || 'no match found'}" (ID: ${targetDestination?.id || 'N/A'})`);

              // Validate the destination ID exists in our cache
              if (targetDestination) {
                const validDestination = viatorDestinations.find(d => d.id === targetDestination.id || d.destinationId === targetDestination.id);
                if (!validDestination) {
                  console.log(`⚠️ Destination ID ${targetDestination.id} not found in cached data, clearing target`);
                  targetDestination = null;
                }
              }
            } else {
              console.log(`❌ No destination context available for search`);
            }

            if (targetDestination) {
              console.log(`🎯 TARGETING ACTIVITIES FOR: ${targetDestination.name} (ID: ${targetDestination.id})`);

              // 🔒 CRITICAL FIX: Store destination context in updatedPreferences immediately
              updatedPreferences = {
                ...updatedPreferences,
                destination: targetDestination.name,
                destinationId: targetDestination.id
              };
              console.log(`🔒 PRESERVED destination context: ${targetDestination.name} (ID: ${targetDestination.id})`);

              // Clear any contaminated cache before searching
              viatorService.clearContaminatedCache();

              // Enhanced search with strict destination filtering
              let searchResults;
              try {
                // Step 1: Try to find matching tags for user interests
                console.log(`🏷️ Attempting tag-based search for interests: "${searchQuery}" in ${targetDestination.name}`);
                const tagResult = await tagManager.findMatchingTags(searchQuery);

                if (tagResult.tagIds.length > 0 && tagResult.confidence > 50) {
                  console.log(`✅ Found ${tagResult.tagIds.length} matching tags with ${tagResult.confidence}% confidence`);
                  console.log(`🏷️ Matched tags: ${tagResult.matchedTags.map(t => t.tagName).join(', ')}`);

                  // Use tag-based search as primary method with strict destination filtering
                  const tagBasedActivities = await enhancedViatorService.searchActivitiesWithTags({
                    destinationId: targetDestination.id,
                    tagIds: tagResult.tagIds,
                    currency,
                    count: 10
                  });

                  if (tagBasedActivities.length > 0) {
                    console.log(`✅ Tag-based search returned ${tagBasedActivities.length} activities for ${targetDestination.name}`);
                    recommendations = tagBasedActivities;
                  } else {
                    console.log(`⚠️ Tag-based search returned no results, falling back to keyword search`);
                    // Fallback to keyword-based search with strict destination enforcement
                    searchResults = await enhancedViatorService.enhancedSearch({
                      keywords: searchQuery,
                      destination: targetDestination.name,
                      destinationId: targetDestination.id,
                      currency,
                      maxResults: 10,
                      enforceDestination: true
                    });
                    recommendations = searchResults.activities;
                  }
                } else {
                  console.log(`⚠️ No suitable tags found (confidence: ${tagResult.confidence}%), using keyword search`);
                  // Fallback to keyword-based search with strict destination enforcement
                  searchResults = await enhancedViatorService.enhancedSearch({
                    keywords: searchQuery,
                    destination: targetDestination.name,
                    destinationId: targetDestination.id,
                    currency,
                    maxResults: 10,
                    enforceDestination: true
                  });
                  recommendations = searchResults.activities;
                }
              } catch (tagError) {
                console.error('❌ Tag-based search failed, falling back to keywords:', tagError);
                // Fallback to existing keyword-based search
                searchResults = await enhancedViatorService.enhancedSearch({
                  keywords: searchQuery,
                  destination: targetDestination.name,
                  currency,
                  maxResults: 10
                });
                recommendations = searchResults.activities;
              }

            // If no results yet, use Ultimate Search Engine
            if (!recommendations || recommendations.length === 0) {
              // Use Ultimate Search Engine for all other categories
              const generalRequest = {
                query: searchQuery,
                destinationId: targetDestination?.id, // Use validated destination ID
                currency: currency || 'USD',
                maxResults: 8,
                userPreferences: {
                  budget: extractBudgetPreferences(message, currency)?.max > 100 ? 'luxury' : 'mid',
                  duration: extractDurationPreferences(message)?.max > 360 ? 'full_day' : 'medium',
                  interests: [searchQuery]
                }
              };

              if (!generalRequest.destinationId) {
                console.log('⚠️ Cannot perform Ultimate Search without a valid destination ID.');
              } else {
                console.log(`🚀 ULTIMATE ENGINE REQUEST:`, generalRequest);
                try {
                  const generalResults = await ultimateViatorEngine.search(generalRequest);
                  recommendations = generalResults.activities;
                  console.log(`🚀 ULTIMATE ENGINE: Found ${recommendations.length} activities.`);
                } catch (ultimateError) {
                  console.error('❌ Ultimate Engine search failed:', ultimateError);
                }
              }
            }


            if (recommendations && recommendations.length > 0) {
              console.log(`✅ Found ${recommendations.length} activities.`);
            } else {
              console.log('⚠️ No activities found through enhanced search, falling back to freetext.');
              // Fallback to freetext search if no results
              const searchResults = await enhancedViatorService.searchProducts(searchQuery, currency);
              let filteredResults = searchResults.slice(0, 10);

              // Apply destination filtering if we identified a target destination
              if (targetDestination) {
                const targetCity = targetDestination.name.toLowerCase().split(',')[0].trim();
                filteredResults = filteredResults.filter(activity => {
                  const location = activity.location.toLowerCase();
                  return location.includes(targetCity);
                });
                console.log(`🔧 Filtered freetext results to ${filteredResults.length} for ${targetDestination.name}`);
              }
              recommendations = filteredResults;
            }

            shouldShowRecommendations = true;
            console.log('✅ Found recommendations:', recommendations?.length || 0);
          }
        } catch (error) {
          console.error('Error fetching recommendations:', error);
        }
      }
      
      // Note: aiResponse is not available in this endpoint scope
      // shouldShowRecommendations is determined by the search logic above

      // No message truncation - activity cards now appear in side panel

      res.json({
        recommendations: recommendations || [],
        shouldShowRecommendations,
        sessionId: conversation.sessionId,
        destination: updatedPreferences?.destination || null,
        searchQuery: updatedPreferences?.lastSearchQuery || null,
      });
    } catch (error) {
      console.error('Error processing recommendations:', error);
      res.status(500).json({ error: 'Failed to process recommendations' });
    }
  });
